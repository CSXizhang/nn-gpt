def drop_conv3x3_block(in_channels, out_channels, stride=1, padding=1, bias=False, dropout_prob=0.0):
    return nn.Sequential(
        nn.Conv2d(in_channels, out_channels, kernel_size=3, stride=stride, padding=padding, bias=bias),
        nn.BatchNorm2d(out_channels),
        nn.SiLU(inplace=True),
        nn.Dropout2d(p=dropout_prob) if dropout_prob > 0 else nn.Identity()
    )
</block>

def __init__(self, in_shape: tuple, out_shape: tuple, prm: dict, device: torch.device) -> None:
    super().__init__()
    self.device = device
    self.use_amp = prm.get("use_amp", False)
    c_in = in_shape[1] if len(in_shape) == 4 else in_shape[0]
    self._input_spec = (c_in, ) + in_shape[2:]
    self.backbone_a = TorchVision(model='shufflenet_v2_x2_0', in_channels=c_in)
    self.backbone_b = TorchVision(model='shufflenet_v2_x1_5', in_channels=c_in)
    self.features = nn.Sequential(FractalUnit(in_channels=c_in, out_channels=24, num_columns=2, loc_drop_prob=0.1, dropout_prob=0.1), FractalUnit(in_channels=24, out_channels=48, num_columns=2, loc_drop_prob=0.1, dropout_prob=0.1))
    self.infer_dimensions_dynamically(out_shape[0])
    self._scaler = GradScaler("cuda", enabled=self.use_amp)

def forward(self, x: torch.Tensor, is_probing: bool = False) -> torch.Tensor:
    x_a = adaptive_pool_flatten(self.backbone_a(x))
    x_b = adaptive_pool_flatten(self.backbone_b(x))
    x_f = adaptive_pool_flatten(self.features(x))
    x = torch.cat([x_a, x_f, x_b], dim=1)
    return self.classifier(x)
</forward>

### Explanation
The 'A_to_Fractal_plus_B' pattern is implemented directly in the forward function. The features are concatenated in the order A, Fractal, B, where A and B are backbones and Fractal is the FractalUnit block. The drop_conv3x3_block function is used to construct the block components. The '_scaler' is used for Automatic Mixed Precision (AMP) training.
