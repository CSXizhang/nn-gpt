def drop_conv3x3_block(in_channels, out_channels, stride=1, padding=1, bias=False, dropout_prob=0.0):
    return nn.Sequential(
        nn.Conv2d(in_channels, out_channels, kernel_size=3, stride=stride, padding=padding, bias=bias),
        nn.BatchNorm2d(out_channels),
        nn.SiLU(),
        nn.Dropout(p=dropout_prob),
        nn.Conv2d(out_channels, out_channels, kernel_size=3, stride=stride, padding=padding, bias=bias),
        nn.BatchNorm2d(out_channels),
        nn.SiLU(),
        nn.Dropout(p=dropout_prob)
    )
</block>

def __init__(self, in_shape: tuple, out_shape: tuple, prm: dict, device: torch.device) -> None:
    super().__init__()
    self.device = device
    self.use_amp = prm.get("use_amp", False)
    c_in = in_shape[1] if len(in_shape) == 4 else in_shape[0]
    self.pattern = 'Fractal_to_DualBackbone'
    self.backbone_a = TorchVision(model='densenet121', in_channels=c_in)
    self.backbone_b = TorchVision(model='densenet169', in_channels=c_in)
    self.features = nn.Sequential(
        FractalUnit(in_channels=c_in, out_channels=128, num_columns=4, loc_drop_prob=0.1, dropout_prob=0.1),
        FractalUnit(in_channels=128, out_channels=256, num_columns=4, loc_drop_prob=0.1, dropout_prob=0.1)
    )
    self._input_spec = in_shape
    self.infer_dimensions_dynamically(num_classes=out_shape[0])
    self._scaler = make_scaler(enabled=self.use_amp)
</init>

def forward(self, x: torch.Tensor, is_probing: bool = False) -> torch.Tensor:
    x = self.backbone_a(x) if is_probing else self.features(x)
    x = self.backbone_b(x) if not is_probing else x
    return x
