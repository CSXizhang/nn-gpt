def drop_conv3x3_block(in_channels, out_channels, stride=1, padding=1, bias=False, dropout_prob=0.0):
    return nn.Sequential(
    nn.Conv2d(in_channels, out_channels, kernel_size=3, stride=stride, padding=padding, bias=bias),
    nn.BatchNorm2d(out_channels),
    nn.ReLU(inplace=True),
    nn.Dropout2d(p=dropout_prob) if dropout_prob > 0 else nn.Identity(),
    nn.GELU()
)

def __init__(self, in_shape: tuple, out_shape: tuple, prm: dict, device: torch.device) -> None:
    super().__init__()
    self.device = device
    self.pattern = 'Fractal_to_DualBackbone'
    self.use_amp = prm.get('use_amp', False)
    dropout_prob = float(prm.get('dropout', 0.1))
    if len(in_shape) == 4:
        c, h, w = (in_shape[1], in_shape[2], in_shape[3])
    else:
        c, h, w = (in_shape[0], in_shape[1], in_shape[2])
    self._input_spec = (c, h, w)
    self.backbone_a = TorchVision('regnet_y_3_2gf', in_channels=c).to(device)
    self.backbone_b = TorchVision('resnet50', in_channels=c).to(device)
    self.features = nn.Sequential()
    curr_ch = c
    channels = [64 * 2 ** i for i in range(1)]
    for i, out_ch in enumerate(channels):
        self.features.add_module(f'unit{i + 1}', FractalUnit(curr_ch, out_ch, 2, 0.15, dropout_prob))
        curr_ch = out_ch
    self.infer_dimensions_dynamically(out_shape[0])
    if self.use_amp:
        self.scaler = GradScaler('cuda', enabled=True)
</init>

def forward(self, x: torch.Tensor, is_probing: bool=False) -> torch.Tensor:
    x = self._norm4d(x).to(self.device)
    x_f_map = self.features(x)
    x_f_img = self._feature_to_input_image(x_f_map, 'feat').to(self.device)
    x_a = self.backbone_a(x_f_img)
    x_b = self.backbone_b(x_f_img)
    x_f = adaptive_pool_flatten(x_f_map)
    fused = torch.cat([x_f, x_a.flatten(1), x_b.flatten(1)], dim=1)
    if is_probing:
        return fused
    return self.classifier(fused)
